CREACIÓN DE TABLAS

CREATE TABLE EMPRESAS (
    ID_EMP VARCHAR(10) PRIMARY KEY,
    NOM_EMP VARCHAR(50) NOT NULL,
    CIU_EMP VARCHAR(50) NOT NULL
);

CREATE TABLE HOTELES (
    ID_HOT VARCHAR(10) PRIMARY KEY,
    NOM_HOT VARCHAR(50) NOT NULL,
    CIU_HOT VARCHAR(50) NOT NULL
);

CREATE TABLE RECEPCIONISTAS (
    ID_REC VARCHAR(10) PRIMARY KEY,
    NOM_REC VARCHAR(50) NOT NULL,
    APE_REC VARCHAR(50) NOT NULL,
    ID_HOT_REC VARCHAR(10) NOT NULL REFERENCES HOTELES(ID_HOT)
);

CREATE TABLE HABITACIONES (
    ID_HAB VARCHAR(10) PRIMARY KEY,
    COD_CUARTO VARCHAR(10) NOT NULL,
    TIP_HAB VARCHAR(20) NOT NULL,
    COS_BASE NUMBER NOT NULL,
    ID_HOT_HAB VARCHAR(10) NOT NULL REFERENCES HOTELES(ID_HOT)
);

CREATE TABLE EJECUTIVOS (
    ID_EJE VARCHAR(10) PRIMARY KEY,
    CED_EJE VARCHAR(10) UNIQUE NOT NULL,
    NOM_EJE VARCHAR(50) NOT NULL,
    APE_EJE VARCHAR(50) NOT NULL,
    SUE_EJE NUMBER NOT NULL,
    ID_EMP_EJE VARCHAR(10) NOT NULL REFERENCES EMPRESAS(ID_EMP),
    ID_SUP_EJE VARCHAR(10) REFERENCES EJECUTIVOS(ID_EJE)
);

CREATE TABLE HOSPEDAJES (
    ID_HOS VARCHAR(10) PRIMARY KEY,
    FEC_ING DATE NOT NULL,
    FEC_SAL DATE NOT NULL,
    COS_APL NUMBER NOT NULL,
    ID_EJE_HOS VARCHAR(10) NOT NULL REFERENCES EJECUTIVOS(ID_EJE),
    ID_HAB_HOS VARCHAR(10) NOT NULL REFERENCES HABITACIONES(ID_HAB),
    ID_REC_HOS VARCHAR(10) NOT NULL REFERENCES RECEPCIONISTAS(ID_REC)
);

CREATE TABLE NOVEDADES (
    ID_NOV VARCHAR(10) PRIMARY KEY,
    DES_NOV VARCHAR(200) NOT NULL,
    FEC_REP DATE NOT NULL,
    ID_HOS_NOV VARCHAR(10) NOT NULL REFERENCES HOSPEDAJES(ID_HOS)
);

SELECT E.NOM_EJE, E.APE_EJE
FROM EJECUTIVOS E
JOIN EMPRESAS EMP ON E.ID_EMP_EJE = EMP.ID_EMP
WHERE EMP.NOM_EMP = 'PETROECUADOR'
AND E.ID_EJE IN (
    SELECT ID_SUP_EJE
    FROM EJECUTIVOS
    WHERE ID_SUP_EJE IS NOT NULL
    GROUP BY ID_SUP_EJE
    HAVING COUNT(*) = (
        SELECT MAX(COUNT(*))
        FROM EJECUTIVOS
        WHERE ID_SUP_EJE IS NOT NULL
        GROUP BY ID_SUP_EJE
    )
)
AND EXISTS (
    SELECT * FROM EJECUTIVOS SUB
    JOIN HOSPEDAJES H ON SUB.ID_EJE = H.ID_EJE_HOS
    JOIN HABITACIONES HAB ON H.ID_HAB_HOS = HAB.ID_HAB
    JOIN HOTELES HOT ON HAB.ID_HOT_HAB = HOT.ID_HOT
    WHERE SUB.ID_SUP_EJE = E.ID_EJE
    AND HOT.NOM_HOT = 'COLON'
    AND EXTRACT(MONTH FROM H.FEC_ING) = 3 
    AND EXTRACT(YEAR FROM H.FEC_ING) = 2026
);

SELECT R.*
FROM RECEPCIONISTAS R
JOIN HOSPEDAJES H ON R.ID_REC = H.ID_REC_HOS
WHERE H.COS_APL = (
    SELECT MAX(H2.COS_APL)
    FROM HOSPEDAJES H2
    JOIN EJECUTIVOS E ON H2.ID_EJE_HOS = E.ID_EJE
    WHERE E.SUE_EJE > (
        SELECT AVG(E3.SUE_EJE)
        FROM EJECUTIVOS E3
        WHERE E3.ID_EMP_EJE = E.ID_EMP_EJE
    )
);

SELECT EMP.ID_EMP, EMP.NOM_EMP, EMP.CIU_EMP
FROM EMPRESAS EMP
JOIN EJECUTIVOS E ON EMP.ID_EMP = E.ID_EMP_EJE
JOIN HOSPEDAJES H ON E.ID_EJE = H.ID_EJE_HOS
JOIN HABITACIONES HAB ON H.ID_HAB_HOS = HAB.ID_HAB
WHERE HAB.TIP_HAB = 'Doble'
AND H.FEC_ING BETWEEN TO_DATE('2026-01-01', 'YYYY-MM-DD') AND TO_DATE('2026-12-31', 'YYYY-MM-DD')
GROUP BY EMP.ID_EMP, EMP.NOM_EMP, EMP.CIU_EMP
HAVING COUNT(H.ID_HOS) > (
    SELECT COUNT(H2.ID_HOS)
    FROM EMPRESAS EMP2
    JOIN EJECUTIVOS E2 ON EMP2.ID_EMP = E2.ID_EMP_EJE
    JOIN HOSPEDAJES H2 ON E2.ID_EJE = H2.ID_EJE_HOS
    JOIN HABITACIONES HAB2 ON H2.ID_HAB_HOS = HAB2.ID_HAB
    WHERE EMP2.NOM_EMP = 'IESS'
    AND HAB2.TIP_HAB = 'Doble'
    AND H2.FEC_ING BETWEEN TO_DATE('2026-01-01', 'YYYY-MM-DD') AND TO_DATE('2026-12-31', 'YYYY-MM-DD')
);

SELECT R.NOM_REC, R.APE_REC
FROM RECEPCIONISTAS R
JOIN HOSPEDAJES H ON R.ID_REC = H.ID_REC_HOS
JOIN EJECUTIVOS E ON H.ID_EJE_HOS = E.ID_EJE
WHERE R.ID_HOT_REC IN (
    SELECT HAB.ID_HOT_HAB
    FROM HABITACIONES HAB
    GROUP BY HAB.ID_HOT_HAB
    -- REQUISITO: COUNT y Agregación (AVG)
    HAVING COUNT(HAB.ID_HAB) > (
        SELECT AVG(CANT_HAB)
        FROM (
            SELECT COUNT(ID_HAB) AS CANT_HAB
            FROM HABITACIONES
            GROUP BY ID_HOT_HAB
        )
    )
)
GROUP BY R.ID_REC, R.NOM_REC, R.APE_REC
HAVING COUNT(DISTINCT E.ID_EMP_EJE) > 1;




-----Preguntas de defensa-----------------------------

1-------------------------------------------------------

En nuestro script original, decidimos usar GROUP BY y HAVING porque es más eficiente. El motor agrupa a todos los supervisores en memoria una sola vez, cuenta sus subordinados y filtra los que coinciden con el MAX(). Es una operación en bloque.

Si tuviéramos que hacerlo con una subconsulta correlacionada, tendríamos que quitar el GROUP BY principal. La consulta evaluaría a cada ejecutivo fila por fila, y por cada uno, ejecutaría un SELECT COUNT para ver cuántos subordinados tiene, comparando ese resultado contra el máximo. Esto carga mucho más al motor porque la subconsulta se ejecuta repetidas veces.

SELECT E.NOM_EJE, E.APE_EJE
FROM EJECUTIVOS E
JOIN EMPRESAS EMP ON E.ID_EMP_EJE = EMP.ID_EMP
WHERE EMP.NOM_EMP = 'PETROECUADOR'
AND (

    SELECT COUNT(*) 
    FROM EJECUTIVOS SUB1 
    WHERE SUB1.ID_SUP_EJE = E.ID_EJE
) = (

    SELECT MAX(COUNT(*)) 
    FROM EJECUTIVOS 
    WHERE ID_SUP_EJE IS NOT NULL 
    GROUP BY ID_SUP_EJE
)
AND EXISTS (
   
    SELECT * FROM EJECUTIVOS SUB2
    JOIN HOSPEDAJES H ON SUB2.ID_EJE = H.ID_EJE_HOS
    JOIN HABITACIONES HAB ON H.ID_HAB_HOS = HAB.ID_HAB
    JOIN HOTELES HOT ON HAB.ID_HOT_HAB = HOT.ID_HOT
    WHERE SUB2.ID_SUP_EJE = E.ID_EJE -- Se correlaciona con el jefe actual
    AND HOT.NOM_HOT = 'COLON'
    AND H.FEC_ING BETWEEN TO_DATE('2026-03-01','YYYY-MM-DD') AND TO_DATE('2026-03-31','YYYY-MM-DD')
);


2-------------------------------------------------------


La subconsulta de comparación salarial es una subconsulta correlacionada ubicada en el WHERE que calcula el promedio de la empresa. Exactamente esta línea:

SELECT AVG(E3.SUE_EJE) FROM EJECUTIVOS E3 WHERE E3.ID_EMP_EJE = E.ID_EMP_EJE

Si intento sombrear solo este pedacito y ejecutarlo, el motor me dará error. Esto ocurre porque la variable E.ID_EMP_EJE depende de la consulta principal (externa). Para mostrar su resultado por separado en vivo, debo 'desvincularla' reemplazando la variable por un valor estático.


SELECT AVG(SUE_EJE) AS PROMEDIO_EMPRESA
FROM EJECUTIVOS 
WHERE ID_EMP_EJE = 'EMP-01';

3-------------------------------------------------------------


Actualmente nuestra consulta usa > ALL (que matemáticamente es lo mismo que > MAX). Esto significa que el ejecutivo debe ganar más que todos y cada uno de los subordinados del 1810. Es una restricción muy fuerte.

Si cambiamos el operador a > ANY, la lógica matemática cambia por completo. Ahora significa 'ganar más que al menos uno de los subordinados'. Esto es matemáticamente equivalente a decir que solo necesita ganar más que el empleado con el sueldo más bajo (> MIN). La consulta se vuelve más permisiva y devolverá más resultados.

SELECT DISTINCT E.NOM_EJE, E.APE_EJE, E.SUE_EJE
FROM EJECUTIVOS E
JOIN HOSPEDAJES H ON E.ID_EJE = H.ID_EJE_HOS
JOIN HABITACIONES HAB ON H.ID_HAB_HOS = HAB.ID_HAB
WHERE UPPER(HAB.TIP_HAB) = 'SUITE'
AND E.SUE_EJE > ANY (
    SELECT SUB.SUE_EJE
    FROM EJECUTIVOS SUB
    JOIN EJECUTIVOS JEFE ON SUB.ID_SUP_EJE = JEFE.ID_EJE
    WHERE JEFE.CED_EJE = '1810'

En la consulta 6 dependemos matemáticamente de calcular el promedio general de habitaciones que tienen los hoteles. Para que este cálculo sea exacto y sin ambigüedades, es obligatorio que el atributo ID_HOT_HAB (la clave foránea en la tabla HABITACIONES) tenga la restricción NOT NULL.

Si permitiéramos valores nulos en esa clave foránea, un usuario podría insertar 'habitaciones fantasma' que no pertenecen a ningún hotel. Cuando el motor ejecute el GROUP BY ID_HOT_HAB, agrupará todas esas habitaciones huérfanas en un 'grupo nulo' contándolo como si fuera un hotel más. Esto destruiría el divisor al calcular el promedio (AVG), dándonos un número irreal que arruinaría la condición de toda la consulta.
